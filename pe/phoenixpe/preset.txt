# Lazarus PE preset: what Apply-HelixPreset.ps1 switches on or off in
# PhoenixPE. Anything not listed keeps PhoenixPE's default, which is already a
# good lean base (Explorer + StartAllBack, networking, audio, ramdisk, 7-Zip,
# Firefox, ISO output).
#
# Format: on|off  <script path under Projects\>   # why
#         set  <script path> | <option> | <value>   # as if picked in PEBakery
# Paths, options and dropdown choices are checked against your PhoenixPE
# release; anything renamed is reported, never guessed at. {addon} is the
# Helix Boot add-on folder (Projects\MyApps\Helix Boot).

# Storage & network drivers: the #1 reason a rescue PE "can't see the disk"
on   PhoenixPE\Drivers\IntelRSTDriver.script                  # NVMe behind Intel VMD/RST (11th gen+ laptops)
on   PhoenixPE\Drivers\NetworkDrivers.script                  # Windows' own extra Wi-Fi/Ethernet drivers

# Tools a technician expects
on   PhoenixPE\Components\WindowsPowerShell.script            # PowerShell (repair scripts, Get-Disk, etc.)
on   PhoenixPE\Components\TaskMgr.script                      # full Task Manager instead of the WinPE one
on   PhoenixPE\Components\Visual C++ Runtime\350-VCRedist-14.script  # so portable apps from USB:\Apps run

# Needs Microsoft's MDOP media, which the build halts without
off  PhoenixPE\Applications\Diagnostics & Repair\MS-DaRT.script

# Comes from USB:\Apps instead (always current, no rebuild needed)
off  PhoenixPE\Applications\Productivity\Notepad++.script

# Our launcher
on   MyApps\Helix Boot\HelixBoot.script

# Lazarus PE look: the phoenix wallpaper and profile picture, dark mode, Seafoam Teal accent
set  PhoenixPE\Tweaks\Wallpaper.script | rbtn_Custom         | True
set  PhoenixPE\Tweaks\Wallpaper.script | fb_CustomImage      | {addon}\wallpaper.jpg
set  PhoenixPE\Tweaks\Wallpaper.script | cmb_Position        | Fill
set  PhoenixPE\Tweaks\Wallpaper.script | cmb_BGcolor         | Black
set  PhoenixPE\Tweaks\Wallpaper.script | cb_ChangeUserProfilePic | True
set  PhoenixPE\Tweaks\Wallpaper.script | fb_UserProfilePic   | {addon}\profile.png
set  PhoenixPE\Tweaks\Theme.script     | cmb_SystemThemeMode | Dark
set  PhoenixPE\Tweaks\Theme.script     | cmb_AppThemeMode    | Dark
set  PhoenixPE\Tweaks\Theme.script     | cmb_ColorProfile    | Seafoam Teal
